**Step 1: Create the auto-merge workflow**

Create `.github/workflows/auto-merge.yml`:

```yaml
on:
  pull_request_target:
    types: [labeled]

permissions:
  contents: write
  pull-requests: write

jobs:
  auto-merge:
    if: github.event.label.name == 'approved'
    runs-on: ubuntu-latest
    steps:
      - name: Check PR only touches catalog/
        id: check
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          PR_NUMBER: ${{ github.event.pull_request.number }}
        run: |
          files=$(gh pr diff "$PR_NUMBER" --name-only)
          non_catalog=$(echo "$files" | grep -v '^catalog/' || true)
          if [ -n "$non_catalog" ]; then
            echo "PR touches non-catalog files — skipping auto-merge:"
            echo "$non_catalog"
            echo "catalog_only=false" >> "$GITHUB_OUTPUT"
          else
            echo "catalog_only=true" >> "$GITHUB_OUTPUT"
          fi

      - name: Enable auto-merge
        if: steps.check.outputs.catalog_only == 'true'
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
          PR_NUMBER: ${{ github.event.pull_request.number }}
        run: |
          gh pr merge "$PR_NUMBER" --squash --auto
          echo "Auto-merge enabled for PR #$PR_NUMBER"
```

**Step 1: Write the backfill script**

Create `scripts/backfill_dates.py`:

```python
# /// script
# requires-python = ">=3.11"
# dependencies = [
#     "python-frontmatter>=1.1.0",
# ]
# ///
"""Backfill created/updated dates into catalog frontmatter from git history.

Usage:
    uv run scripts/backfill_dates.py              # dry-run (print changes)
    uv run scripts/backfill_dates.py --write      # write changes to files
"""

from __future__ import annotations

import subprocess
import sys
from pathlib import Path

import frontmatter

ROOT = Path(__file__).resolve().parent.parent
CATALOG_DIR = ROOT / "catalog"


def git_created(path: Path) -> str:
    """Get the date this file was first added to git."""
    result = subprocess.run(
        ["git", "log", "--follow", "--diff-filter=A", "--format=%aI", "--", str(path)],
        capture_output=True, text=True, cwd=ROOT,
    )
    dates = result.stdout.strip().splitlines()
    if not dates:
        # File not yet committed — use today
        from datetime import date
        return date.today().isoformat()
    # --follow returns oldest last
    return dates[-1][:10]


def git_updated(path: Path) -> str:
    """Get the date of the most recent commit touching this file."""
    result = subprocess.run(
        ["git", "log", "-1", "--format=%aI", "--", str(path)],
        capture_output=True, text=True, cwd=ROOT,
    )
    date_str = result.stdout.strip()
    if not date_str:
        from datetime import date
        return date.today().isoformat()
    return date_str[:10]


def backfill(write: bool = False) -> None:
    changed = 0
    skipped = 0

    for subdir in ["mappings", "frames", "categories"]:
        dir_path = CATALOG_DIR / subdir
        for md_file in sorted(dir_path.glob("*.md")):
            post = frontmatter.load(md_file)
            modified = False

            if "created" not in post.metadata:
                post.metadata["created"] = git_created(md_file)
                modified = True

            if "updated" not in post.metadata:
                post.metadata["updated"] = git_updated(md_file)
                modified = True

            if modified:
                changed += 1
                print(f"  {subdir}/{md_file.name}: created={post.metadata['created']} updated={post.metadata['updated']}")
                if write:
                    frontmatter.dump(post, md_file)
            else:
                skipped += 1

    print(f"\n{changed} files {'updated' if write else 'would be updated'}, {skipped} already had dates.")
    if not write and changed > 0:
        print("Run with --write to apply changes.")


def main() -> None:
    write = "--write" in sys.argv
    backfill(write=write)


if __name__ == "__main__":
    main()
```

```python
# /// script
# requires-python = ">=3.11"
# dependencies = [
#     "python-frontmatter>=1.1.0",
# ]
# ///
"""Generate daily or weekly digest reports for Metaphorex.

Usage:
    uv run scripts/digest.py daily                # today's digest
    uv run scripts/digest.py daily --date 2026-03-12  # specific date
    uv run scripts/digest.py weekly               # this week's digest
    uv run scripts/digest.py weekly --date 2026-03-12  # week containing date
"""

from __future__ import annotations

import argparse
import json
import subprocess
import sys
from datetime import date, datetime, timedelta
from pathlib import Path

import frontmatter

ROOT = Path(__file__).resolve().parent.parent
CATALOG_DIR = ROOT / "catalog"
DIGESTS_DIR = ROOT / "docs" / "digests"
REPO = "metaphorex/metaphorex"


def gh(*args: str) -> str:
    """Run a gh CLI command and return stdout."""
    result = subprocess.run(
        ["gh", *args],
        capture_output=True, text=True, cwd=ROOT,
    )
    if result.returncode != 0:
        print(f"gh command failed: gh {' '.join(args)}", file=sys.stderr)
        print(result.stderr, file=sys.stderr)
    return result.stdout.strip()


def catalog_counts() -> dict[str, int]:
    """Count entries in each catalog subdirectory."""
    counts = {}
    for subdir in ["mappings", "frames", "categories"]:
        dir_path = CATALOG_DIR / subdir
        counts[subdir] = len(list(dir_path.glob("*.md")))
    return counts


def merged_prs(since: str, until: str) -> list[dict]:
    """Get PRs merged in a date range."""
    raw = gh(
        "pr", "list", "-R", REPO,
        "--state", "merged",
        "--search", f"merged:{since}..{until}",
        "--limit", "100",
        "--json", "number,title,mergedAt,files",
    )
    if not raw:
        return []
    return json.loads(raw)


def new_mappings_from_prs(prs: list[dict]) -> list[str]:
    """Extract mapping slugs from merged PRs that touched catalog/mappings/."""
    slugs = []
    for pr in prs:
        for f in pr.get("files", []):
            path = f.get("path", "")
            if path.startswith("catalog/mappings/") and path.endswith(".md"):
                slug = path.removeprefix("catalog/mappings/").removesuffix(".md")
                if f.get("additions", 0) > 0:
                    slugs.append(slug)
    return sorted(set(slugs))


def stats_from_issues(since: str) -> list[dict]:
    """Parse ## stats: lines from issue comments since a date."""
    # Get comments from import-project issues
    issues_raw = gh(
        "issue", "list", "-R", REPO,
        "--label", "import-project",
        "--state", "all",
        "--limit", "50",
        "--json", "number",
    )
    if not issues_raw:
        return []

    issues = json.loads(issues_raw)
    all_stats = []

    for issue in issues:
        comments_raw = gh(
            "api", f"repos/{REPO}/issues/{issue['number']}/comments",
            "--paginate", "--jq",
            f'[.[] | select(.created_at >= "{since}") | .body] | join("\\n")',
        )
        if not comments_raw:
            continue

        # Parse stats lines (reuse logic from stats.py)
        import re
        pattern = re.compile(
            r"^## stats:(?P<agent>[a-z-]+):(?P<model>[a-z0-9.-]+)\s+(?P<kv>.+)$"
        )
        kv_pattern = re.compile(r"(?P<key>[a-z_]+)=(?P<value>[^\s]+)")

        for line in comments_raw.splitlines():
            m = pattern.match(line.strip())
            if not m:
                continue
            kvs = {km.group("key"): km.group("value") for km in kv_pattern.finditer(m.group("kv"))}
            tokens_in = int(kvs.get("tokens_in", 0))
            tokens_out = int(kvs.get("tokens_out", 0))
            usd_in = float(kvs.get("usd_in_per_mtok", 0))
            usd_out = float(kvs.get("usd_out_per_mtok", 0))
            cost = tokens_in * usd_in / 1_000_000 + tokens_out * usd_out / 1_000_000
            all_stats.append({
                "agent": m.group("agent"),
                "model": m.group("model"),
                "tokens_in": tokens_in,
                "tokens_out": tokens_out,
                "cost": cost,
            })

    return all_stats


def aggregate_stats(stats: list[dict]) -> str:
    """Format stats into a markdown table."""
    if not stats:
        return "No agent activity recorded.\n"

    by_agent: dict[str, dict] = {}
    for s in stats:
        key = f"{s['agent']}|{s['model']}"
        if key not in by_agent:
            by_agent[key] = {"agent": s["agent"], "model": s["model"],
                             "runs": 0, "tokens": 0, "cost": 0.0}
        by_agent[key]["runs"] += 1
        by_agent[key]["tokens"] += s["tokens_in"] + s["tokens_out"]
        by_agent[key]["cost"] += s["cost"]

    lines = []
    lines.append("| Agent | Model | Runs | Tokens | Cost |")
    lines.append("|-------|-------|------|--------|------|")
    total_runs = total_tokens = 0
    total_cost = 0.0
    for row in sorted(by_agent.values(), key=lambda r: -r["cost"]):
        lines.append(
            f"| {row['agent']} | {row['model']} | {row['runs']} "
            f"| {row['tokens']:,} | ${row['cost']:.2f} |"
        )
        total_runs += row["runs"]
        total_tokens += row["tokens"]
        total_cost += row["cost"]
    lines.append(
        f"| **Total** | | **{total_runs}** "
        f"| **{total_tokens:,}** | **${total_cost:.2f}** |"
    )
    return "\n".join(lines) + "\n"


def pipeline_status() -> str:
    """Get import project completion status."""
    raw = gh(
        "issue", "list", "-R", REPO,
        "--label", "import-project",
        "--state", "open",
        "--limit", "20",
        "--json", "number,title",
    )
    if not raw:
        return "No active import projects.\n"

    projects = json.loads(raw)
    lines = []
    lines.append("| Project | Progress | Status |")
    lines.append("|---------|----------|--------|")

    for proj in projects:
        # Query sub-issues
        sub_raw = gh(
            "api", "graphql",
            "-f", f"owner=metaphorex", "-f", f"name=metaphorex",
            "-F", f"number={proj['number']}",
            "-f", "query=query($owner: String!, $name: String!, $number: Int!) { repository(owner: $owner, name: $name) { issue(number: $number) { subIssues(first: 100) { nodes { state } } } } }",
            "--jq", ".data.repository.issue.subIssues.nodes",
        )
        if not sub_raw:
            lines.append(f"| {proj['title'][:40]} | ? | unknown |")
            continue

        subs = json.loads(sub_raw)
        total = len(subs)
        closed = sum(1 for s in subs if s["state"] == "CLOSED")
        if total == 0:
            lines.append(f"| {proj['title'][:40]} | 0/0 | no sub-issues |")
        else:
            pct = closed / total * 100
            status = "complete" if closed == total else "active"
            lines.append(f"| {proj['title'][:40]} | {closed}/{total} ({pct:.0f}%) | {status} |")

    return "\n".join(lines) + "\n"


def kaizen_backlog() -> str:
    """List open kaizen issues."""
    raw = gh(
        "issue", "list", "-R", REPO,
        "--label", "kaizen:pipeline,kaizen:content",
        "--state", "open",
        "--limit", "20",
        "--json", "number,title,createdAt",
    )
    if not raw:
        return "No open kaizen issues.\n"

    issues = json.loads(raw)
    if not issues:
        return "No open kaizen issues.\n"

    lines = []
    for issue in issues:
        created = issue["createdAt"][:10]
        lines.append(f"- #{issue['number']}: {issue['title']} (opened {created})")
    return "\n".join(lines) + "\n"


def generate_daily(target_date: date) -> str:
    """Generate a daily digest for a specific date."""
    since = target_date.isoformat()
    until = (target_date + timedelta(days=1)).isoformat()
    counts = catalog_counts()
    prs = merged_prs(since, until)
    new_slugs = new_mappings_from_prs(prs)
    stats = stats_from_issues(since)

    lines = []
    lines.append("---")
    lines.append(f"date: {target_date.isoformat()}")
    lines.append("type: daily")
    lines.append("---")
    lines.append("")
    lines.append(f"# Daily Digest — {target_date.isoformat()}")
    lines.append("")

    # New content
    lines.append("## New Content")
    if new_slugs:
        lines.append(f"- {len(new_slugs)} new mappings merged")
        for slug in new_slugs:
            lines.append(f"  - [{slug}](https://metaphorex.org/mappings/{slug}/)")
    else:
        lines.append("- No new mappings merged today")
    lines.append(f"- Catalog totals: **{counts['mappings']}** mappings, "
                 f"**{counts['frames']}** frames, **{counts['categories']}** categories")
    lines.append("")

    # Costs
    lines.append("## Costs")
    lines.append(aggregate_stats(stats))

    # Pipeline
    lines.append("## Pipeline Status")
    lines.append(pipeline_status())

    # Kaizen
    lines.append("## Kaizen Backlog")
    lines.append(kaizen_backlog())

    return "\n".join(lines)


def generate_weekly(target_date: date) -> str:
    """Generate a weekly digest for the week containing target_date."""
    # ISO week: Monday = 0
    monday = target_date - timedelta(days=target_date.weekday())
    sunday = monday + timedelta(days=6)
    iso_year, iso_week, _ = monday.isocalendar()
    since = monday.isoformat()
    until = (sunday + timedelta(days=1)).isoformat()
    counts = catalog_counts()
    prs = merged_prs(since, until)
    new_slugs = new_mappings_from_prs(prs)
    stats = stats_from_issues(since)

    total_cost = sum(s["cost"] for s in stats)

    lines = []
    lines.append("---")
    lines.append(f"date: {monday.isoformat()}")
    lines.append("type: weekly")
    lines.append(f"week: {iso_year}-W{iso_week:02d}")
    lines.append("---")
    lines.append("")
    lines.append(f"# Weekly Digest — Week {iso_week}, {iso_year}")
    lines.append("")

    # Growth
    lines.append("## Growth")
    lines.append(f"- {len(new_slugs)} new mappings merged this week")
    lines.append(f"- Catalog totals: **{counts['mappings']}** mappings, "
                 f"**{counts['frames']}** frames, **{counts['categories']}** categories")
    lines.append("")

    # Costs
    lines.append("## Costs (7-day)")
    lines.append(aggregate_stats(stats))

    # Pipeline
    lines.append("## Pipeline Health")
    lines.append(pipeline_status())

    # Kaizen
    lines.append("## Kaizen Backlog")
    lines.append(kaizen_backlog())

    # Highlights
    lines.append("## Highlights")
    lines.append("")
    lines.append("<!-- Edit this section before posting to LinkedIn -->")
    lines.append("")
    if new_slugs[:3]:
        for slug in new_slugs[:3]:
            lines.append(f"- [{slug}](https://metaphorex.org/mappings/{slug}/)")
    else:
        lines.append("- (no new content this week)")
    lines.append("")

    return "\n".join(lines)


def main() -> None:
    parser = argparse.ArgumentParser(description="Metaphorex digest generator")
    sub = parser.add_subparsers(dest="command", required=True)

    daily_parser = sub.add_parser("daily", help="Generate daily digest")
    daily_parser.add_argument("--date", help="Date (YYYY-MM-DD), default today")

    weekly_parser = sub.add_parser("weekly", help="Generate weekly digest")
    weekly_parser.add_argument("--date", help="Date in target week (YYYY-MM-DD), default today")

    args = parser.parse_args()

    if args.command == "daily":
        target = date.fromisoformat(args.date) if args.date else date.today()
        content = generate_daily(target)
        out_path = DIGESTS_DIR / f"{target.isoformat()}.md"
        out_path.parent.mkdir(parents=True, exist_ok=True)
        out_path.write_text(content)
        print(f"Daily digest written to {out_path}")
        print(content)

    elif args.command == "weekly":
        target = date.fromisoformat(args.date) if args.date else date.today()
        monday = target - timedelta(days=target.weekday())
        iso_year, iso_week, _ = monday.isocalendar()
        content = generate_weekly(target)
        out_dir = DIGESTS_DIR / "weekly"
        out_dir.mkdir(parents=True, exist_ok=True)
        out_path = out_dir / f"{iso_year}-W{iso_week:02d}.md"
        out_path.write_text(content)
        print(f"Weekly digest written to {out_path}")
        print(content)


if __name__ == "__main__":
    main()
```

**Step 1: Create hardlinks for digest content**

The site uses hardlinks from `site/src/content/` to `catalog/`. Digests
live in `docs/digests/`, so we need the same pattern:

```bash
ln -f docs/digests/*.md site/src/content/digests/ 2>/dev/null || true
```

Actually, this won't work for CI — hardlinks don't survive git clone. The
Astro glob loader can reference paths outside `src/content/` using relative
paths. Let's use that approach instead.
